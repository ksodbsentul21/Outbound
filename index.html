const SHEET_NAME = 'Outbound'; const VALIDATION_SHEET = 'Validation'; /* ========================================================== WAJIB DIISI ========================================================== Masukkan ID folder Google Drive untuk menyimpan Foto Driver. Contoh URL folder: https://drive.google.com/drive/folders/1ABCxxxxxxxxxxxx Yang dimasukkan hanya: 1ABCxxxxxxxxxxxx */ const FOTO_FOLDER_ID = 'PASTE_ID_FOLDER_FOTO_DI_SINI'; function doGet(e) { try { const action = e && e.parameter ? e.parameter.action : ''; if (action === 'getVendors') { return jsonResponse( getVendors() ); } return jsonResponse({ success: true, message: 'H JOEL LOGISTICS API aktif.' }); } catch (error) { return jsonResponse({ success: false, message: error.message }); } } function doPost(e) { try { if ( !e || !e.postData || !e.postData.contents ) { throw new Error( 'Data POST tidak ditemukan.' ); } const data = JSON.parse( e.postData.contents ); if ( data.action === 'save' ) { return jsonResponse( saveOutbound(data) ); } throw new Error( 'Action tidak dikenal.' ); } catch (error) { return jsonResponse({ success: false, message: error.message }); } } function jsonResponse(data) { return ContentService .createTextOutput( JSON.stringify(data) ) .setMimeType( ContentService.MimeType.JSON ); } function getVendors() { const ss = SpreadsheetApp .getActiveSpreadsheet(); const sheet = ss.getSheetByName( VALIDATION_SHEET ); if (!sheet) { return { success: false, message: 'Sheet Validation tidak ditemukan.', vendors: [] }; } const lastRow = sheet.getLastRow(); if (lastRow < 2) { return { success: true, vendors: [] }; } /* ========================================================== DAFTAR VENDOR DIAMBIL DARI KOLOM A MULAI BARIS 2 ========================================================== */ let vendors = sheet .getRange( 2, 1, lastRow - 1, 1 ) .getDisplayValues() .flat() .map( value => String(value).trim() ) .filter(Boolean); /* Hilangkan vendor duplikat. */ vendors = [...new Set(vendors)]; return { success: true, vendors: vendors }; } function saveOutbound(data) { const tanggalPickup = String( data.tanggalPickup || '' ).trim(); const namaDriver = String( data.namaDriver || '' ).trim(); const platNo = String( data.platNo || '' ) .trim() .toUpperCase(); const vendor = String( data.vendor || '' ).trim(); let awbs = Array.isArray(data.awbs) ? data.awbs : []; awbs = awbs .map( awb => String(awb).trim() ) .filter(Boolean); /* Hilangkan AWB duplikat. */ awbs = [...new Set(awbs)]; if (!tanggalPickup) { throw new Error( 'Tanggal Pick Up wajib diisi.' ); } if (!namaDriver) { throw new Error( 'Nama Driver wajib diisi.' ); } if (!platNo) { throw new Error( 'Plat No wajib diisi.' ); } if (!vendor) { throw new Error( 'Vendor wajib dipilih.' ); } if ( awbs.length === 0 ) { throw new Error( 'Belum ada AWB.' ); } if ( !data.fotoDriver ) { throw new Error( 'Foto Driver wajib diambil.' ); } /* ========================================================== AMBIL SHEET OUTBOUND ========================================================== */ const ss = SpreadsheetApp .getActiveSpreadsheet(); let sheet = ss.getSheetByName( SHEET_NAME ); /* Kalau sheet Outbound belum ada, otomatis dibuat. */ if (!sheet) { sheet = ss.insertSheet( SHEET_NAME ); sheet.getRange( 1, 1, 1, 7 ).setValues([ [ 'Tanggal Pick Up', 'Timestamp', 'AWB', 'Nama Driver', 'Plat No', 'Vendor', 'Foto Driver' ] ]); } /* ========================================================== UPLOAD FOTO DRIVER ========================================================== */ const fotoUrl = saveDriverPhoto( data.fotoDriver, namaDriver, platNo ); /* ========================================================== TANGGAL PICK UP ========================================================== */ const pickupParts = tanggalPickup.split('-'); const pickupDate = new Date( Number(pickupParts[0]), Number(pickupParts[1]) - 1, Number(pickupParts[2]) ); /* ========================================================== TIMESTAMP ========================================================== */ const timestamp = new Date(); /* ========================================================== BUAT BARIS ========================================================== */ const rows = awbs.map( awb => [ pickupDate, timestamp, awb, namaDriver, platNo, vendor, fotoUrl ] ); const startRow = sheet.getLastRow() + 1; /* Tulis semua AWB sekaligus. */ sheet .getRange( startRow, 1, rows.length, 7 ) .setValues(rows); /* ========================================================== FORMAT ========================================================== */ sheet .getRange( startRow, 1, rows.length, 1 ) .setNumberFormat( 'dd/MM/yyyy' ); sheet .getRange( startRow, 2, rows.length, 1 ) .setNumberFormat( 'dd/MM/yyyy HH:mm:ss' ); /* AWB dibuat TEXT agar angka panjang tidak berubah menjadi notasi ilmiah. */ sheet .getRange( startRow, 3, rows.length, 1 ) .setNumberFormat('@'); /* Plat nomor dibuat TEXT. */ sheet .getRange( startRow, 5, rows.length, 1 ) .setNumberFormat('@'); return { success: true, jumlah: rows.length, message: rows.length + ' AWB berhasil disimpan.', fotoUrl: fotoUrl }; } function saveDriverPhoto( dataUrl, namaDriver, platNo ) { if ( !FOTO_FOLDER_ID || FOTO_FOLDER_ID === 'PASTE_ID_FOLDER_FOTO_DI_SINI' ) { throw new Error( 'ID Folder Foto Driver belum dimasukkan di Kode.gs.' ); } /* Ambil folder Drive berdasarkan ID. */ const folder = DriveApp.getFolderById( FOTO_FOLDER_ID ); /* Pisahkan header Base64. */ const parts = String(dataUrl) .split(','); if ( parts.length < 2 ) { throw new Error( 'Format Foto Driver tidak valid.' ); } const header = parts[0]; const base64 = parts[1]; const mimeMatch = header.match( /data:(.*?);base64/ ); const mimeType = mimeMatch ? mimeMatch[1] : 'image/jpeg'; const bytes = Utilities .base64Decode( base64 ); /* Bersihkan nama driver untuk dijadikan nama file. */ const safeDriver = namaDriver .replace( /[^a-zA-Z0-9\s_-]/g, '' ) .replace( /\s+/g, '_' ); const safePlat = platNo .replace( /[^a-zA-Z0-9]/g, '_' ); const timestamp = Utilities.formatDate( new Date(), Session.getScriptTimeZone(), 'yyyyMMdd_HHmmss' ); const fileName = 'DRIVER_' + safeDriver + '_' + safePlat + '_' + timestamp + '.jpg'; const blob = Utilities.newBlob( bytes, mimeType, fileName ); const file = folder.createFile( blob ); /* Mencoba membuat link foto dapat dilihat melalui link. */ try { file.setSharing( DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW ); } catch (error) { /* Jika Google Workspace melarang public sharing, file tetap tersimpan. */ console.log( 'Sharing foto tidak dapat diubah: ' + error.message ); } return file.getUrl(); }
